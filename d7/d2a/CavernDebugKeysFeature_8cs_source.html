247 #region Cursor Hiding
248
249 // We hide the cursor when it's not moving.
250 // We use coroutines instead of an update loop because most of the time the mouse isn't going to be moving
251 // And this saves on compute cost in that case (although it's slightly worse if the mouse is moving often)
252 IEnumerator hideMouseCoroutine = null;
253 IEnumerator HideMouse()
254 {
255 yield return new WaitForSeconds(3); // after three seconds, hide the mouse
256 Cursor.visible = false;
257 }
258
259 void OnMouseMove(InputAction.CallbackContext context)
260 {
261 cavernSetup.StopCoroutine(hideMouseCoroutine);
262
263 Cursor.visible = true;
264 hideMouseCoroutine = HideMouse();
265 cavernSetup.StartCoroutine(hideMouseCoroutine);
266 }
267 #endregion
