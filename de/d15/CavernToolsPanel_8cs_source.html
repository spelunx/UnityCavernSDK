84 private void AddRendererToURPAsset()
85 {
86
87 UniversalRenderPipelineAsset urpAsset = AssetDatabase.LoadAssetAtPath<UniversalRenderPipelineAsset>("Assets/Settings/PC_RPAsset.asset");
88 ScriptableRendererData newRenderer = AssetDatabase.LoadAssetAtPath<ScriptableRendererData>("Packages/com.spelunx.cavern.sdk/Runtime/Scripts/Camera/Cavern Renderer.asset");
89
90 if (urpAsset == null || newRenderer == null)
91 {
92 Debug.LogError("Failed to add Cavern Renderer to URP asset. Please do this manually.");
93 return;
94 }
95
96 SerializedObject serializedAsset = new SerializedObject(urpAsset);
97 SerializedProperty rendererListProperty = serializedAsset.FindProperty("m_RendererDataList");
98
99 if (rendererListProperty != null && rendererListProperty.isArray)
100 {
101 // Check if the renderer exists already in the array
102 for(int i = 0; i < rendererListProperty.arraySize; i++)
103 {
104 SerializedProperty el = rendererListProperty.GetArrayElementAtIndex(i);
105 if(el.objectReferenceValue == newRenderer)
106 {
107 // already exists, so we stop here
108 return;
109 }
110 }
111
112 // Add the renderer to the end of the array
113 int newIndex = rendererListProperty.arraySize;
114 rendererListProperty.InsertArrayElementAtIndex(newIndex);
115
116 SerializedProperty element = rendererListProperty.GetArrayElementAtIndex(newIndex);
117 element.objectReferenceValue = newRenderer;
118
119 // 4. Apply the modified properties
120 serializedAsset.ApplyModifiedProperties();
121
122 // 5. Save the asset
123 EditorUtility.SetDirty(urpAsset);
124 AssetDatabase.SaveAssets();
125 }
126 else
127 {
128 Debug.LogError("Failed to add Cavern Renderer to URP asset. Please do this manually.");
129 }
130 }
