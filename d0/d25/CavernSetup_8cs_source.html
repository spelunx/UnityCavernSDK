150 // generates a mesh used for the preview or other uses
151 public Mesh GenerateMesh()
152 {
153 Mesh mesh = new Mesh();
154 // Have about one panel every 10 degrees. A reasonable number.
155 int numPanels = Mathf.Max(1, (int)(cavernAngle / 10.0f));
156 int numVertices = (numPanels + 1) * 2;
157
158 Vector3[] positions = new Vector3[numVertices];
159 Vector3[] normals = new Vector3[numVertices];
160 Vector2[] uvs = new Vector2[numVertices];
161 int[] indices = new int[numPanels * 6];
162
163 /********************************************** Generate inner surface. **********************************************/
164
165 float cavernBottomHeight = cavernElevation;
166 float cavernTopHeight = cavernHeight + cavernElevation;
167
168 //float topUV = (previewEye == PreviewEye.Left) ? 1.0f : 0.5f;
169 //float bottomUV = (previewEye == PreviewEye.Left) ? 0.5f : 0.0f;
170 float topUV = 1.0f;
171 float bottomUV = 0.5f;
172
173 float deltaAngle = cavernAngle / (float)numPanels;
174
175 // Create vertices of surface.
176 for (int i = 0; i <= numPanels; i++)
177 {
178 float ratio = (float)i / (float)numPanels;
179 float currAngle = (ratio - 0.5f) * cavernAngle;
180
181 // Take note that angle 0 points down the Z-axis, not the X-axis.
182 float directionX = Mathf.Sin(currAngle * Mathf.Deg2Rad);
183 float directionZ = Mathf.Cos(currAngle * Mathf.Deg2Rad);
184
185 positions[i * 2] = new Vector3(cavernRadius * directionX, cavernTopHeight, cavernRadius * directionZ); // Top vertex.
186 normals[i * 2] = new Vector3(cavernRadius * directionX, 0.0f, cavernRadius * directionZ);
187 uvs[i * 2] = new Vector2((float)i / (float)numPanels, topUV);
188
189 positions[i * 2 + 1] = new Vector3(cavernRadius * directionX, cavernBottomHeight, cavernRadius * directionZ); // Top vertex.
190 normals[i * 2 + 1] = new Vector3(cavernRadius * directionX, 0.0f, cavernRadius * directionZ);
191 uvs[i * 2 + 1] = new Vector2((float)i / (float)numPanels, bottomUV);
192 }
193
194 // Assign indices of each panel.
195 // Each panel is a quad made up of 2 triangles.
196 // Unity uses a CLOCKWISE WINDING ORDER for its triangles.
197 for (int i = 0; i < numPanels; ++i)
198 {
199 // Triangle 1
200 indices[i * 6] = i * 2;
201 indices[i * 6 + 1] = i * 2 + 2;
202 indices[i * 6 + 2] = i * 2 + 1;
203
204 // Triangle 2
205 indices[i * 6 + 3] = i * 2 + 1;
206 indices[i * 6 + 4] = i * 2 + 2;
207 indices[i * 6 + 5] = i * 2 + 3;
208 }
209
210 mesh.name = "Cavern Mesh";
211 mesh.vertices = positions;
212 mesh.normals = normals;
213 mesh.uv = uvs;
214 mesh.triangles = indices;
215 return mesh;
216 }
217 }
